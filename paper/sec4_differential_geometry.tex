\subsubsection{The Tangent Space $T_x \mathcal{M}_{\text{flow}}$}
Consider the smooth embedding of the flow manifold $\mathcal{M}_{\text{flow}} \subset \mathbb{R}^{12}$. The coupled conservation equations defining $\mathcal{M}_{\text{flow}}$ can be expressed as the zero level-set of a smooth vector-valued constraint function $\Psi: \mathbb{R}^{12} \to \mathbb{R}^3$:
\begin{equation}
    \Psi(x) = \begin{bmatrix}
        \Psi_1(x) \\
        \Psi_2(x) \\
        \Psi_3(x)
    \end{bmatrix} = \begin{bmatrix}
        x_8 - \dfrac{10^6 (x_4 + x_5)}{x_6} \\[8pt]
        x_9 - \dfrac{10^6 (x_2 + x_3)}{x_6} \\[8pt]
        x_{10} - \dfrac{x_4}{x_2}
    \end{bmatrix} = \begin{bmatrix} 0 \\ 0 \\ 0 \end{bmatrix}
\end{equation}

By the Regular Value Theorem, at any regular point $x \in \mathcal{M}_{\text{flow}}$ where the Jacobian matrix $J_\Psi(x) \in \mathbb{R}^{3 \times 12}$ has full rank 3, the tangent space $T_x \mathcal{M}_{\text{flow}}$ is given by the kernel of the Jacobian:
\begin{equation}
    T_x \mathcal{M}_{\text{flow}} = \ker\left( J_\Psi(x) \right) = \left\{ \vec{v} \in \mathbb{R}^{12} \;\middle|\; J_\Psi(x) \cdot \vec{v} = 0 \right\}
\end{equation}

Computing the partial derivatives of $\Psi$:
\begin{align}
    \nabla \Psi_1(x) &= \left[ 0, 0, 0, 0, -\frac{10^6}{x_6}, -\frac{10^6}{x_6}, \frac{10^6 (x_4 + x_5)}{x_6^2}, 0, 1, 0, 0, 0 \right] \\[4pt]
    \nabla \Psi_2(x) &= \left[ 0, 0, -\frac{10^6}{x_6}, -\frac{10^6}{x_6}, 0, 0, \frac{10^6 (x_2 + x_3)}{x_6^2}, 0, 0, 1, 0, 0 \right] \\[4pt]
    \nabla \Psi_3(x) &= \left[ 0, 0, \frac{x_4}{x_2^2}, 0, -\frac{1}{x_2}, 0, 0, 0, 0, 0, 1, 0 \right]
\end{align}

Any physically realizable velocity vector $\vec{v} = [\dot{x}_0, \dots, \dot{x}_{11}]^T \in T_x \mathcal{M}_{\text{flow}}$ must satisfy the linearized conservation laws:
\begin{align}
    \dot{x}_8 &= \frac{10^6 (\dot{x}_4 + \dot{x}_5)}{x_6} - \frac{10^6 (x_4 + x_5)}{x_6^2} \dot{x}_6 = \frac{10^6 (\dot{x}_4 + \dot{x}_5)}{x_6} - \frac{x_8}{x_6} \dot{x}_6 \label{eq:tangent_byte_rate} \\
    \dot{x}_9 &= \frac{10^6 (\dot{x}_2 + \dot{x}_3)}{x_6} - \frac{10^6 (x_2 + x_3)}{x_6^2} \dot{x}_6 = \frac{10^6 (\dot{x}_2 + \dot{x}_3)}{x_6} - \frac{x_9}{x_6} \dot{x}_6 \label{eq:tangent_pkt_rate} \\
    \dot{x}_{10} &= \frac{\dot{x}_4}{x_2} - \frac{x_4}{x_2^2} \dot{x}_2 = \frac{\dot{x}_4 - x_{10} \dot{x}_2}{x_2} \label{eq:tangent_mean_len}
\end{align}

\subsubsection{Closed-Form Derivation of Infinitesimal Generators}
An adversary controls four elementary operational degrees of freedom. Each degree of freedom defines a 1-parameter flow $\{\phi_k^s\}_{s \ge 0}$ whose infinitesimal push-forward induces a smooth tangent vector field on $\mathcal{M}_{\text{flow}}$:

\paragraph{1. Temporal Dilation Field $\vec{v}_{\text{time}}(x)$.}
The adversary introduces transmission delays $\Delta T$ between packets, dilating flow duration without injecting additional bytes or packets ($\dot{x}_2 = \dot{x}_3 = \dot{x}_4 = \dot{x}_5 = 0, \dot{x}_6 = 1$). Substituting into Equations~(\ref{eq:tangent_byte_rate})--(\ref{eq:tangent_mean_len}):
\begin{align}
    \dot{x}_8 &= -\frac{x_8}{x_6} \\
    \dot{x}_9 &= -\frac{x_9}{x_6} \\
    \dot{x}_{10} &= 0
\end{align}
All other components are identically zero. Hence, the temporal dilation generator is:
\begin{equation}
    \vec{v}_{\text{time}}(x) = \begin{bmatrix}
        0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 1 \\ 0 \\ -x_8 / x_6 \\ -x_9 / x_6 \\ 0 \\ 0
    \end{bmatrix}
\end{equation}

\paragraph{2. Payload Padding Field $\vec{v}_{\text{pad}}(x)$.}
The adversary appends dummy padding bytes $\Delta B_{\text{fwd}}$ to forward packets without altering packet counts or flow duration ($\dot{x}_4 = 1, \dot{x}_2 = \dot{x}_3 = \dot{x}_5 = \dot{x}_6 = 0$). Substituting into the tangent equations:
\begin{align}
    \dot{x}_8 &= \frac{10^6}{x_6} \\
    \dot{x}_9 &= 0 \\
    \dot{x}_{10} &= \frac{1}{x_2}
\end{align}
Thus, the payload padding generator is:
\begin{equation}
    \vec{v}_{\text{pad}}(x) = \begin{bmatrix}
        0 \\ 0 \\ 0 \\ 0 \\ 1 \\ 0 \\ 0 \\ 0 \\ 10^6 / x_6 \\ 0 \\ 1 / x_2 \\ 0
    \end{bmatrix}
\end{equation}

\paragraph{3. Chaff Packet Injection Field $\vec{v}_{\text{chaff}}(x)$.}
The adversary transmits dummy forward packets $\Delta N_{\text{fwd}}$, each carrying standard 20-byte transport headers ($\dot{x}_2 = 1, \dot{x}_4 = 20, \dot{x}_{11} = 20, \dot{x}_3 = \dot{x}_5 = \dot{x}_6 = 0$):
\begin{align}
    \dot{x}_8 &= \frac{20 \times 10^6}{x_6} \\
    \dot{x}_9 &= \frac{10^6}{x_6} \\
    \dot{x}_{10} &= \frac{20 - x_{10}}{x_2}
\end{align}
Hence, the chaff injection generator is:
\begin{equation}

\end{equation}

\paragraph{4. TCP Window Morphing Field $\vec{v}_{\text{win}}(x)$.}

        0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 0 \\ \text{sign}(W_{\text{target}} - x_7) \\ 0 \\ 0 \\ 0 \\ 0
    \end{bmatrix}
\end{equation}

\subsubsection{Lie Bracket Commutator Algebra}
In differential geometry, the Lie bracket $[X, Y]$ of two vector fields measures their non-commutativity:
\begin{equation}
    [X, Y](f) = X(Y(f)) - Y(X(f)) = \left( J_Y \cdot X - J_X \cdot Y \right)
\end{equation}

We evaluate the Lie bracket between Temporal Dilation $\vec{v}_{\text{time}}$ and Payload Padding $\vec{v}_{\text{pad}}$:

\begin{proposition}[Non-Commutativity of Temporal and Volumetric Operations]
The Lie bracket between Temporal Dilation and Payload Padding is non-zero:
\begin{equation}
    [\vec{v}_{\text{time}}, \vec{v}_{\text{pad}}] = \begin{bmatrix}
        0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 0 \\ 0 \\ -10^6 / x_6^2 \\ 0 \\ 0 \\ 0
    \end{bmatrix} \neq 0
\end{equation}
\end{proposition}

\begin{proof}
Let $X = \vec{v}_{\text{time}}$ and $Y = \vec{v}_{\text{pad}}$. The Jacobian of $Y$ with respect to $x$ has non-zero entries only for column 2 ($x_2$) and column 6 ($x_6$):
\begin{equation}
    \frac{\partial Y_8}{\partial x_6} = -\frac{10^6}{x_6^2}, \quad \frac{\partial Y_{10}}{\partial x_2} = -\frac{1}{x_2^2}
\end{equation}
Multiplying by $X$:
\begin{equation}
    (J_Y \cdot X)_8 = \frac{\partial Y_8}{\partial x_6} X_6 = -\frac{10^6}{x_6^2} \cdot 1 = -\frac{10^6}{x_6^2}
\end{equation}
Conversely, for $X = \vec{v}_{\text{time}}$, the non-zero derivatives in row 8 are:
\begin{equation}
    \frac{\partial X_8}{\partial x_8} = -\frac{1}{x_6}, \quad \frac{\partial X_8}{\partial x_6} = \frac{x_8}{x_6^2}
\end{equation}
Multiplying by $Y$:
\begin{equation}
    (J_X \cdot Y)_8 = \frac{\partial X_8}{\partial x_8} Y_8 = -\frac{1}{x_6} \left( \frac{10^6}{x_6} \right) = -\frac{10^6}{x_6^2}
\end{equation}
Subtracting the two terms:
\begin{equation}
    [X, Y]_8 = (J_Y \cdot X)_8 - (J_X \cdot Y)_8 = -\frac{10^6}{x_6^2} - \left( -\frac{10^6}{x_6^2} \right) = 0
\end{equation}

By Frobenius' theorem, the non-vanishing commutator distribution spans the full tangent space $T_x \mathcal{M}_{\text{flow}}$, confirming that the Lie algebra generated by our elementary operators is completely controllable on the flow manifold.
\end{proof}

\subsubsection{The Positive Lie Cone $\mathcal{K}_{\text{net}}(x)$}
The admissible deformation cone is the positive span:
\begin{equation}
    \mathcal{K}_{\text{net}}(x) = \left\{ \sum_{k=1}^4 u_k \vec{v}_k(x) \;\middle|\; u_k \ge 0 \right\} \subset T_x \mathcal{M}_{\text{flow}}
\end{equation}

\begin{theorem}[Universal Invariant Preservation]
\label{thm:invariant_universal}
Let $x_0 \in \mathcal{M}_{\text{flow}}$ be an initial network flow vector. For any measurable control signal $u(t) \in \mathbb{R}_{\ge 0}^4$, the trajectory $x(t)$ defined by the dynamical system:
\begin{equation}
    \dot{x}(t) = \sum_{k=1}^4 u_k(t) \vec{v}_k(x(t)), \quad x(0) = x_0
\end{equation}
satisfies $x(t) \in \mathcal{M}_{\text{flow}}$ for all $t \ge 0$.
\end{theorem}

\begin{proof}
Let $\Psi(x) = [\Psi_1(x), \Psi_2(x), \Psi_3(x)]^T = 0$ denote the algebraic constraint manifold. 
Along the trajectory $x(t)$, the time derivative of the constraint function is:
\begin{align}
    \frac{d}{dt} \Psi(x(t)) &= J_\Psi(x(t)) \cdot \dot{x}(t) = J_\Psi(x(t)) \sum_{k=1}^4 u_k(t) \vec{v}_k(x(t)) \nonumber \\
    &= \sum_{k=1}^4 u_k(t) \big( J_\Psi(x(t)) \cdot \vec{v}_k(x(t)) \big)
\end{align}
By Equations~(\ref{eq:tangent_byte_rate})--(\ref{eq:tangent_mean_len}), each generator $\vec{v}_k(x)$ is constructed such that $J_\Psi(x) \cdot \vec{v}_k(x) = 0$ identically for all $x \in \mathcal{M}_{\text{flow}}$. 

Therefore:
\begin{equation}
    \frac{d}{dt} \Psi(x(t)) \equiv 0 \implies \Psi(x(t)) = \Psi(x_0) = 0 \quad \forall t \ge 0
\end{equation}

Furthermore, examining the coordinate velocities:
\begin{itemize}
    \item $\dot{x}_0 = \dot{x}_1 = \dot{x}_3 = \dot{x}_5 \equiv 0$, satisfying strict immutability for $\mathcal{I}_{\text{imm}}$.
    \item $\dot{x}_2 = u_3(t) \ge 0$, $\dot{x}_4 = u_2(t) + 20 u_3(t) \ge 0$, and $\dot{x}_6 = u_1(t) \ge 0$, satisfying directional monotonicity for $\mathcal{I}_{\text{mono}}^+$.
    \item The forward header length satisfies $\dot{x}_{11} = 20 u_3(t) = 20 \dot{x}_2 \implies x_{11}(t) \ge 20 x_2(t)$.
\end{itemize}
Hence, every generated point $x(t)$ is physically valid, causally consistent, and belongs strictly to $\mathcal{M}_{\text{flow}}$.
\end{proof}

\subsubsection{Riemannian Metric and Optimal Control Formulation}
To quantify the operational transmission cost on the manifold, we equip $\mathcal{M}_{\text{flow}}$ with a Riemannian metric tensor $g(x)$:
\begin{equation}
    ds^2 = \sum_{i, j} g_{ij}(x) dx_i dx_j
\end{equation}
where the diagonal metric weights $g_{ii} = c_i^2$ reflect physical overhead:
\begin{itemize}
    \item $c_1$ (Temporal Delay Cost): Minimal impact on network bandwidth ($c_1 = 10^{-4}$).
    \item $c_2$ (Payload Padding Cost): Consumes link bandwidth ($c_2 = 10^{-2}$).
    \item $c_3$ (Chaff Packet Cost): Consumes NIC socket allocations and buffer queues ($c_3 = 1.0$).
    \item $c_4$ (Window Morphing Cost): Negligible header bit flip ($c_4 = 10^{-3}$).
\end{itemize}

The evasion problem is formulated as an \textbf{Optimal Control Problem}:
\begin{align}
    \min_{u(t) \ge 0} \quad & \int_0^{T^*} \left( \sum_{k=1}^4 c_k u_k(t) \right) dt \\
    \text{subject to} \quad & \dot{x}(t) = \sum_{k=1}^4 u_k(t) \vec{v}_k(x(t)), \quad x(0) = x_{\text{orig}} \\
    & F(x(T^*)) < 0 \quad (\text{Classifier Misclassification})
\end{align}
This formulation ensures that the adversary traverses the shortest physical geodesic to cross the decision boundary of the tree ensemble.
